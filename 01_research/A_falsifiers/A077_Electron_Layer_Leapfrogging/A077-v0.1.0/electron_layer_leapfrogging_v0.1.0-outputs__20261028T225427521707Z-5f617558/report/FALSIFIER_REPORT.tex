\section{Exact mathematical equations and observables}
\paragraph{E1: two-ring interaction.} Let source ring $j$ be
$\mathbf X_j(\varphi)=(R_j\cos\varphi,R_j\sin\varphi,z_j)$, evaluated at $\mathbf x_i=(R_i,0,z_i)$. For $i\neq j$,
\begin{equation}
\mathbf u_{i\leftarrow j}=\frac{\Gamma_j}{4\pi}\int_0^{2\pi}
 \frac{\mathbf X'_j(\varphi)\times(\mathbf x_i-\mathbf X_j(\varphi))}
 {\left(\lVert\mathbf x_i-\mathbf X_j(\varphi)\rVert^2+a_j^2\right)^{3/2}}\,d\varphi.
\end{equation}
This is a \emph{specified softened kernel}, not the exact Euler Green's function of a resolved core. Velocity dimensions: $[\Gamma][\mathbf X'][\mathbf r]/[r]^3=L/T$.
\paragraph{E2: self translation (closure).}
\begin{equation}
 U_i^{\rm self}=\frac{\Gamma_i}{4\pi R_i}\left[\log\left(\frac{8R_i}{a_i}\right)-\frac14\right],\quad
 \dot R_i=\sum_{j\ne i}u_{r,i\leftarrow j},\quad
 \dot z_i=U_i^{\rm self}+\sum_{j\ne i}u_{z,i\leftarrow j}.
\end{equation}
The $1/4$ coefficient is a conventional model choice, not a fitted SST electron constant. The coupling of a soft mutual kernel and a slender-core self formula is an approximate composite closure.
\paragraph{E3: material-volume proxy and E4: order events.}
\begin{equation}
 a_i^2(t)R_i(t)=a_i^2(0)R_i(0),\qquad
 \Delta z(t)=z_2(t)-z_1(t),\qquad N_{\rm swap}=\#\{t:\Delta z(t)=0\text{ with a sign change}\}.
\end{equation}
$[a^2R]=L^3$, $[\Delta z]=L$ and $N_{\rm swap}$ is dimensionless. In particular $N_{\rm swap}$ is NOT an electron orbital principal quantum number.
\paragraph{Numerical scheme.} Source angle midpoint/equispaced trapezoidal sum with $n_\varphi$ points; classical fixed-step fourth-order Runge--Kutta for the reduced ordinary differential equations. No grid-based fluid conservation, finite-core Euler convergence or strain diagnostic can be inferred from these ODE tests.
