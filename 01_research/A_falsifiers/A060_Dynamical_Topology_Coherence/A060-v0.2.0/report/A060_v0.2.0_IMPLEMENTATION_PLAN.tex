\section{M01: Landreman exact Euler family}
Use the first exact family of \cite{Landreman2026}, written as a dimensionless Euler field, with $0\leq\varepsilon<1$,
\begin{gather}

 F=\sqrt{1-(1-s)^2-4z^2},\\

\end{gather}
The admissible domain has $F>0$, $s>0$. The fraction bars in the original PDF were visually inspected: the denominator $s$ covers the \emph{entire} numerator in the first two components. The complete reference identities are

\end{equation}

\section{M03: VA/VT projector, including Helmholtz closure}
For $|\boldsymbol\omega|>0$ define
\begin{equation}
 \mathbf{VA}=\frac{\boldsymbol\omega^TS\boldsymbol\omega}{|\boldsymbol\omega|^2}\boldsymbol\omega,\quad
 \mathbf{VT}=S\boldsymbol\omega-\mathbf{VA},\quad
 \boldsymbol\omega\cdot\mathbf{VT}=0.
\end{equation}
For Chiarini's periodic modified forcing $\nabla^2\mathbf g=\nabla\times\mathbf{VA}$, the correct relation is
\begin{equation}
 \nabla\times\mathbf g=-\mathbb P_\perp\mathbf{VA},\qquad
 \mathbb P_\perp=I-\nabla\nabla^{-2}\nabla\cdot.
\end{equation}
Since $\nabla\cdot\mathbf{VA}$ need not vanish, an unprojected local VA removal is generally \emph{not} equal to the modified momentum equation.
