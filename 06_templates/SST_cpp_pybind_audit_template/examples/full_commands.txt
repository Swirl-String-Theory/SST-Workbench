# ---------------------------------------------------------------------------
# run_example.py — single run
# ---------------------------------------------------------------------------
# Defaults (--a 2 --b 3 --expected auto=a+b)
python run_example.py

# Operands
python run_example.py --a 1.5 --b 2.5
python run_example.py --a 0 --b 0
python run_example.py --a -1 --b 4

# Explicit expected value (pass/fail threshold)
python run_example.py --a 2 --b 3 --expected 5
python run_example.py --a 2 --b 3 --expected 6

# Backend selection / build control
python run_example.py --force-python
python run_example.py --skip-build
python run_example.py --force-build
python run_example.py --build-verbose
python run_example.py --force-build --build-verbose

# Output (bare names go under {folder}_outputs/)
python run_example.py --out example_out.json
python run_example.py --summary-only
python run_example.py --a 2 --b 3 --out example_out.json --summary-only

# Combined (typical audit run)
python run_example.py --a 2 --b 3 --force-build --build-verbose --out example_cpp.json
python run_example.py --a 2 --b 3 --force-python --out example_python.json
