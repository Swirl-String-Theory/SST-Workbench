\section*{Observables and Euler equation}
For incompressible, unforced, inviscid periodic Euler,
\begin{equation}
\nabla\cdot\mathbf{u}=0,\qquad
\partial_t\mathbf{u}=\mathbb{P}(\mathbf{u}\times(\nabla\times\mathbf{u})),
\end{equation}
where $\mathbb P$ is the Leray projector. For a moving material contour $C(t)$,
\begin{equation}
\Gamma(t)=\oint_{C(t)}\mathbf{u}\cdot d\boldsymbol\ell,
\qquad \frac{d\Gamma(t)}{dt}=0
\end{equation}
under sufficiently regular Euler flow. Define the trajectory refinement discrepancy
\begin{equation}
D_{h,h'}(T)=\left[\frac{1}{N_p}\sum_{j=1}^{N_p}
\|\mathbf{X}^{(h)}_j(T)-\mathbf{X}^{(h')}_j(T)\|_{\mathrm{periodic}}^2\right]^{1/2}.
\end{equation}
This quantity has dimensions of length in the model's non-dimensional coordinates. It must not be silently treated as SI length.
