cmake_minimum_required(VERSION 3.18)
project(maxwell_sst_native LANGUAGES CXX)
find_package(Python COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(maxwell_sst_native src/maxwell_sst_native.cpp)
target_compile_features(maxwell_sst_native PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(maxwell_sst_native PRIVATE /O2)
else()
  target_compile_options(maxwell_sst_native PRIVATE -O3)
endif()
