cmake_minimum_required(VERSION 3.16)
project(SSTMaxwellReciprocalFalsifier LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(sst_reciprocal_core cpp/sst_reciprocal_core.cpp)
if(MSVC)
  target_compile_options(sst_reciprocal_core PRIVATE /O2 /W4)
else()
  target_compile_options(sst_reciprocal_core PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
