\section{Exact equations and independent controls}
\paragraph{D1 and D2A (preserved from v0.2.0).} D1 uses exact affine 3D Euler strain and rotating material loops; $\Gamma=2\pi\Omega R^2$ is constant. D2A uses $u_\theta=Cr\exp(-r^2/a^2)$, Euler radial pressure $dp/dr=\rho r\Omega^2$, and independent Cartesian tracer integration. D2A has finite energy per unit axial length, not finite total three-dimensional energy.
\paragraph{D2H --- Hill's spherical vortex.}
In the translating frame ($s^2=x^2+y^2$, $r^2=s^2+z^2$), for $r<a$:
\begin{align}
 u_x&=\frac{3Uxz}{2a^2},&u_y&=\frac{3Uyz}{2a^2},&u_z&=\frac{3U}{2}\left(1-\frac{2s^2+z^2}{a^2}\right),\\
 \boldsymbol\omega&=A_H(-y,x,0),&A_H&=\frac{15U}{2a^2},&\Gamma_H&=\frac{2A_H a^3}{3},\\
 U&=\frac{\Gamma_H}{5a}=\frac{2a^2A_H}{15}.
\end{align}
Outside the sphere, the stationary co-moving flow is uniform $-U\mathbf e_z$ plus a decaying potential dipole:
\begin{equation}
 (u_x,u_y)=\frac{3Ua^3z}{2r^5}(x,y),\qquad u_z=-U+Ua^3\left(r^{-3}-\frac{3s^2}{2r^5}\right).
\end{equation}
With $\chi=(3A_HU/4)s^2(r^2/a^2-1)$ inside and zero outside, $p=\rho[(U^2-|\mathbf u|^2)/2+\chi]$. Independent centered differences test $\nabla\cdot\mathbf u=0$, $\nabla\times\mathbf u=\boldsymbol\omega$, and $\rho(\mathbf u\cdot\nabla)\mathbf u+\nabla p=0$ away from the interface.
\paragraph{D2N --- normalized GP/NLS radial core.}
\begin{align}
 i\partial_T\Psi&=-\frac12\nabla^2\Psi+(|\Psi|^2-1)\Psi,\quad \Psi=f(r)e^{i\theta},\\
 0&=-\frac12\left(f''+f'/r-f/r^2\right)+(f^2-1)f.
\end{align}
Boundary $f(0)=0$ and $f(R_{\rm box})=1$. A numerical boundary-value solution is checked on independent radial grids using a separately differenced differential residual. \emph{Global $U(1)$ phase is not an operational clock}. Gauge zero-mode residual is audited, but no independently stable BdG internal mode is claimed.
\paragraph{D2R --- thin-ring identity.}
\begin{equation}
 E=\frac12\rho\Gamma^2R[\ln(8R/a)-\alpha],\quad P=\rho\Gamma\pi R^2,\quad U=\frac{dE}{dP}=\frac{\Gamma}{4\pi R}[\ln(8R/a)-\beta].
\end{equation}
For constant $a$, $\beta=\alpha-1$; for $a^2R=\mathrm{const}$, $\beta=\alpha-3/2$. Roberts--Grant $\alpha\simeq1.615$ is a reference value. Substitution $\alpha=\varphi$ is a different unverified hypothesis. The SST-69 fat-core estimate $R/a\simeq0.898$ is \emph{outside} the tested thin-ring regime ($R/a=40$).
