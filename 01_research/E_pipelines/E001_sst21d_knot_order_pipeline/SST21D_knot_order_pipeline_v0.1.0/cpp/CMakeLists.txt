cmake_minimum_required(VERSION 3.16)
project(sst21d_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(sst21d_native SHARED src/sst21d_native.cpp)
target_include_directories(sst21d_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(sst21d_native PRIVATE /O2 /W4)
else()
  target_compile_options(sst21d_native PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
set_target_properties(sst21d_native PROPERTIES OUTPUT_NAME "sst21d_native")
